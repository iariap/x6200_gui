cmake_minimum_required(VERSION 3.23)
project(ui_probe C)
set(CMAKE_C_STANDARD 11)
set(LVGL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../.x6100_ref/tools/js8_ui_harness/deps/src/lvgl)
file(GLOB_RECURSE LVGL_SRC ${LVGL_DIR}/src/*.c)
add_library(lvgl STATIC ${LVGL_SRC})
target_include_directories(lvgl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${LVGL_DIR} ${LVGL_DIR}/..)
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE=1)
find_package(PNG REQUIRED)
add_executable(ui_probe probe.c)
target_include_directories(ui_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${LVGL_DIR} ${LVGL_DIR}/..)
target_compile_definitions(ui_probe PRIVATE LV_CONF_INCLUDE_SIMPLE=1)
target_link_libraries(ui_probe PRIVATE lvgl PNG::PNG m)
